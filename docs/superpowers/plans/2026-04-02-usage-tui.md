**Goal:** Add a `tui` subcommand to the usage CLI that renders vertical bar charts (calls, tokens) and a line chart (cache hit %) with adaptive axes, using Rich for terminal rendering.

**Architecture:** New module `src/llm/usage_tui.py` owns all chart rendering logic. `UsageTracker` gets a new `timeseries()` query method that returns per-bucket aggregated rows. The existing `usage_cli.py` gains `tui day|month|week` subcommands that call `timeseries()` then hand data to the TUI renderer. Three charts stacked vertically: calls bar chart, tokens stacked bar chart (input/output), cache hit % line chart.

**Tech Stack:** `rich` (new dependency), Python `datetime`/`zoneinfo` for timezone handling.

async def timeseries(
    self,
    *,
    period: str,
    date: str | None = None,
    tz_offset_hours: float = 0.0,
) -> list[dict[str, Any]]:
    """Return per-bucket aggregated rows.

    period='day'   → date='YYYY-MM-DD', buckets by hour (00-23)
    period='month' → date='YYYY-MM',    buckets by day  (01-31)
    period='week'  → date='YYYY-MM-DD', buckets by day  (MM-DD) for 7 days ending on date
    """
    tz_offset = f"{tz_offset_hours:+.1f} hours"

    if period == "day":
        if date is None:
            date = datetime.now(UTC).strftime("%Y-%m-%d")
        return await self.query_raw(
            """
            SELECT
                strftime('%H', ts, ?) AS bucket,
                COUNT(*)                                    AS calls,
                COALESCE(SUM(input_tokens), 0)              AS input_tokens,
                COALESCE(SUM(cache_read_tokens), 0)         AS cache_read_tokens,
                COALESCE(SUM(cache_create_tokens), 0)       AS cache_create_tokens,
                COALESCE(SUM(output_tokens), 0)             AS output_tokens
            FROM llm_calls
            WHERE strftime('%Y-%m-%d', ts, ?) = ?
            GROUP BY bucket ORDER BY bucket
            """,
            (tz_offset, tz_offset, date),
        )
    elif period == "month":
        if date is None:
            date = datetime.now(UTC).strftime("%Y-%m")
        return await self.query_raw(
            """
            SELECT
                strftime('%d', ts, ?) AS bucket,
                COUNT(*)                                    AS calls,
                COALESCE(SUM(input_tokens), 0)              AS input_tokens,
                COALESCE(SUM(cache_read_tokens), 0)         AS cache_read_tokens,
                COALESCE(SUM(cache_create_tokens), 0)       AS cache_create_tokens,
                COALESCE(SUM(output_tokens), 0)             AS output_tokens
            FROM llm_calls
            WHERE strftime('%Y-%m', ts, ?) = ?
            GROUP BY bucket ORDER BY bucket
            """,
            (tz_offset, tz_offset, date),
        )
    elif period == "week":
        if date is None:
            date = datetime.now(UTC).strftime("%Y-%m-%d")
        return await self.query_raw(
            """
            SELECT
                strftime('%m-%d', ts, ?) AS bucket,
                COUNT(*)                                    AS calls,
                COALESCE(SUM(input_tokens), 0)              AS input_tokens,
                COALESCE(SUM(cache_read_tokens), 0)         AS cache_read_tokens,
                COALESCE(SUM(cache_create_tokens), 0)       AS cache_create_tokens,
                COALESCE(SUM(output_tokens), 0)             AS output_tokens
            FROM llm_calls
            WHERE date(ts, ?) BETWEEN date(?, '-6 days') AND ?
            GROUP BY bucket ORDER BY bucket
            """,
            (tz_offset, tz_offset, date, date),
        )
    else:
        raise ValueError(f"unknown period: {period!r}")
```

"""TUI dashboard: vertical bar charts and line chart using Rich."""

from __future__ import annotations

import math
import os
from datetime import datetime, timezone
from typing import Any

from rich.console import Console
from rich.text import Text


def _nice_ticks(max_val: float, num_ticks: int = 5) -> list[float]:
    """Compute human-friendly evenly-spaced Y axis ticks from 0 to >= max_val."""
    if max_val <= 0:
        max_val = 1.0
    raw_step = max_val / (num_ticks - 1)
    magnitude = 10 ** math.floor(math.log10(raw_step))
    residual = raw_step / magnitude
    if residual <= 1.0:
        nice_step = magnitude
    elif residual <= 2.0:
        nice_step = 2 * magnitude
    elif residual <= 5.0:
        nice_step = 5 * magnitude
    else:
        nice_step = 10 * magnitude
    return [nice_step * i for i in range(num_ticks)]


def _fmt_axis_label(val: float) -> str:
    """Format a numeric value for Y axis display (K/M suffixes)."""
    if val >= 1_000_000:
        return f"{val / 1_000_000:.1f}M"
    if val >= 1_000:
        return f"{val / 1_000:.0f}K"
    if val == int(val):
        return str(int(val))
    return f"{val:.1f}"


def _local_tz_label() -> str:
    """Return local timezone label like 'UTC+8' or 'UTC-5'."""
    now = datetime.now(timezone.utc).astimezone()
    offset = now.utcoffset()
    if offset is None:
        return "UTC"
    total_seconds = int(offset.total_seconds())
    hours, remainder = divmod(abs(total_seconds), 3600)
    sign = "+" if total_seconds >= 0 else "-"
    if remainder:
        minutes = remainder // 60
        return f"UTC{sign}{hours}:{minutes:02d}"
    return f"UTC{sign}{hours}"


def _local_tz_offset_hours() -> float:
    """Return local timezone offset in hours (e.g. 8.0 for UTC+8)."""
    now = datetime.now(timezone.utc).astimezone()
    offset = now.utcoffset()
    if offset is None:
        return 0.0
    return offset.total_seconds() / 3600
```

def render_bar_chart(
    *,
    buckets: list[str],
    values: list[int | float],
    y_label: str,
    chart_height: int = 15,
    chart_width: int = 0,
    bar_style: str = "green",
) -> Text:
    """Render a vertical bar chart as a Rich Text object.

    Uses full-block (█) and half-block (▄) for sub-cell precision.
    Y axis auto-scales with nice ticks. X axis adapts label density
    to fit chart_width.
    """
    if chart_width <= 0:
        chart_width = os.get_terminal_size(fallback=(80, 24)).columns - 2

    max_val = max(values) if values else 0
    ticks = _nice_ticks(max_val, num_ticks=min(chart_height, 6))
    y_max = ticks[-1] if ticks[-1] > 0 else 1.0

    # Y axis label width (widest tick label + 1 for the axis char)
    tick_labels = [_fmt_axis_label(t) for t in ticks]
    y_width = max(len(l) for l in tick_labels) + 2  # padding + "┤" or "┼"

    # Available width for bars
    bar_area = chart_width - y_width
    n = len(buckets)
    if n == 0:
        return Text(f" {y_label}\n (no data)\n")

    # Bar width and gap: each bar gets at least 1 col, prefer 3
    col_width = max(1, bar_area // n)
    bar_w = max(1, col_width - 1)  # 1 col gap

    # Compute bar heights in half-rows (2 * chart_height units)
    half_heights = []
    for v in values:
        if y_max > 0:
            h = v / y_max * chart_height * 2
        else:
            h = 0
        half_heights.append(int(round(h)))

    lines: list[str] = []
    # Y label
    lines.append(f" {y_label}")

    # Render rows top to bottom
    tick_set = {int(round(t / y_max * chart_height * 2)) for t in ticks} if y_max > 0 else set()
    for row in range(chart_height, 0, -1):
        threshold_full = row * 2
        threshold_half = row * 2 - 1

        # Y axis label
        y_val = y_max * row / chart_height
        # Show label only at tick positions
        label = ""
        for t, tl in zip(ticks, tick_labels):
            if abs(t - y_val) < y_max / chart_height / 2:
                label = tl
                break
        prefix = f"{label:>{y_width - 1}} {'┤' if row < chart_height else '┤'}"

        bar_chars = ""
        for i in range(n):
            h = half_heights[i]
            if h >= threshold_full:
                bar_chars += "█" * bar_w
            elif h >= threshold_half:
                bar_chars += "▄" * bar_w
            else:
                bar_chars += " " * bar_w
            if col_width > bar_w:
                bar_chars += " " * (col_width - bar_w)

        lines.append(prefix + bar_chars)

    # X axis line
    x_line = f"{'0':>{y_width - 1}} ┼" + "─" * (col_width * n)
    lines.append(x_line)

    # X axis labels — adapt density to fit
    x_labels = " " * y_width + " "
    step = max(1, n * col_width // (bar_area // max(3, len(buckets[0]) + 1)) if bar_area > 0 else 1)
    step = max(1, step)
    for i in range(n):
        lbl = buckets[i] if i % step == 0 else ""
        x_labels += f"{lbl:<{col_width}}"
    lines.append(x_labels)

    text = Text()
    for i, line in enumerate(lines):
        if i > 0:
            text.append("\n")
        text.append(line)

    return text
```

def render_stacked_bar_chart(
    *,
    buckets: list[str],
    series_a: list[int | float],
    label_a: str,
    series_b: list[int | float],
    label_b: str,
    y_label: str,
    chart_height: int = 15,
    chart_width: int = 0,
    style_a: str = "cyan",
    style_b: str = "yellow",
) -> Text:
    """Render a stacked vertical bar chart (series_a on bottom, series_b on top)."""
    if chart_width <= 0:
        chart_width = os.get_terminal_size(fallback=(80, 24)).columns - 2

    totals = [a + b for a, b in zip(series_a, series_b)]
    max_val = max(totals) if totals else 0
    ticks = _nice_ticks(max_val, num_ticks=min(chart_height, 6))
    y_max = ticks[-1] if ticks[-1] > 0 else 1.0

    tick_labels = [_fmt_axis_label(t) for t in ticks]
    y_width = max(len(l) for l in tick_labels) + 2
    unit = _auto_unit(y_max)

    bar_area = chart_width - y_width
    n = len(buckets)
    if n == 0:
        return Text(f" {y_label}\n (no data)\n")

    col_width = max(1, bar_area // n)
    bar_w = max(1, col_width - 1)

    # Half-heights for each series (a stacked below b)
    half_a = []
    half_b = []
    for a, b in zip(series_a, series_b):
        ha = int(round(a / y_max * chart_height * 2)) if y_max > 0 else 0
        hb = int(round(b / y_max * chart_height * 2)) if y_max > 0 else 0
        half_a.append(ha)
        half_b.append(hb)

    lines: list[str] = []
    # Header with legend
    lines.append(f" {y_label} ({unit})" + " " * 10 + f"█ {label_a}  ░ {label_b}")

    for row in range(chart_height, 0, -1):
        threshold_full = row * 2
        threshold_half = row * 2 - 1

        y_val = y_max * row / chart_height
        label = ""
        for t, tl in zip(ticks, tick_labels):
            if abs(t - y_val) < y_max / chart_height / 2:
                label = tl
                break
        prefix = f"{label:>{y_width - 1}} ┤"

        bar_chars = ""
        for i in range(n):
            total_h = half_a[i] + half_b[i]
            a_h = half_a[i]
            if total_h >= threshold_full:
                # This row is filled — is it in A or B zone?
                if a_h >= threshold_full:
                    bar_chars += "█" * bar_w
                else:
                    bar_chars += "░" * bar_w
            elif total_h >= threshold_half:
                if a_h >= threshold_half:
                    bar_chars += "▄" * bar_w
                else:
                    bar_chars += "░" * bar_w
            else:
                bar_chars += " " * bar_w
            if col_width > bar_w:
                bar_chars += " " * (col_width - bar_w)

        lines.append(prefix + bar_chars)

    x_line = f"{'0':>{y_width - 1}} ┼" + "─" * (col_width * n)
    lines.append(x_line)

    x_labels = " " * y_width + " "
    step = max(1, len(buckets) // (bar_area // max(3, len(buckets[0]) + 1))) if bar_area > 0 else 1
    for i in range(n):
        lbl = buckets[i] if i % step == 0 else ""
        x_labels += f"{lbl:<{col_width}}"
    lines.append(x_labels)

    text = Text()
    for i, line in enumerate(lines):
        if i > 0:
            text.append("\n")
        text.append(line)
    return text


def _auto_unit(max_val: float) -> str:
    """Return the display unit for a max value."""
    if max_val >= 1_000_000:
        return "M"
    if max_val >= 1_000:
        return "K"
    return ""
```

def render_line_chart(
    *,
    buckets: list[str],
    values: list[float | None],
    y_label: str,
    chart_height: int = 8,
    chart_width: int = 0,
    line_style: str = "magenta",
    warn_below: float | None = None,
) -> Text:
    """Render a line chart with adaptive Y range (not starting from 0).

    Uses Unicode drawing characters: · ─ / \\ for the line.
    Y axis range = [min - padding, max + padding], auto-scaled.
    """
    if chart_width <= 0:
        chart_width = os.get_terminal_size(fallback=(80, 24)).columns - 2

    real_values = [v for v in values if v is not None]
    if not real_values:
        return Text(f" {y_label}\n (no data)\n")

    v_min = min(real_values)
    v_max = max(real_values)
    if v_min == v_max:
        # Flat line — create a range around the value
        padding = max(5.0, v_min * 0.1)
        v_min = max(0, v_min - padding)
        v_max = v_max + padding
    else:
        span = v_max - v_min
        padding = span * 0.15
        v_min = max(0, v_min - padding)
        v_max = v_max + padding

    # Nice ticks for the adaptive range
    ticks = _nice_range_ticks(v_min, v_max, num_ticks=min(chart_height, 5))
    y_lo = ticks[0]
    y_hi = ticks[-1]
    y_span = y_hi - y_lo if y_hi > y_lo else 1.0

    tick_labels = [_fmt_axis_label(t) for t in ticks]
    y_width = max(len(l) for l in tick_labels) + 2

    bar_area = chart_width - y_width
    n = len(buckets)
    if n == 0:
        return Text(f" {y_label}\n (no data)\n")

    col_width = max(1, bar_area // n)

    # Map values to row positions (0 = bottom, chart_height-1 = top)
    positions: list[int | None] = []
    for v in values:
        if v is None:
            positions.append(None)
        else:
            row = int(round((v - y_lo) / y_span * (chart_height - 1)))
            row = max(0, min(chart_height - 1, row))
            positions.append(row)

    lines: list[str] = []
    lines.append(f" {y_label}")

    # Build a 2D grid
    grid: list[list[str]] = [[" "] * (col_width * n) for _ in range(chart_height)]

    # Place line segments
    for i in range(n):
        p = positions[i]
        if p is None:
            continue
        # Place point
        x_center = i * col_width + col_width // 2
        if 0 <= x_center < col_width * n:
            grid[p][x_center] = "·"

        # Connect to next point
        if i + 1 < n and positions[i + 1] is not None:
            p_next = positions[i + 1]
            x_start = x_center
            x_end = (i + 1) * col_width + col_width // 2
            _draw_line_segment(grid, x_start, p, x_end, p_next)

    # Render rows top to bottom
    for row in range(chart_height - 1, -1, -1):
        y_val = y_lo + y_span * row / (chart_height - 1)
        label = ""
        for t, tl in zip(ticks, tick_labels):
            if abs(t - y_val) < y_span / chart_height / 2:
                label = tl
                break
        prefix = f"{label:>{y_width - 1}} ┤"
        row_str = "".join(grid[row])
        lines.append(prefix + row_str)

    # X axis
    x_line = f"{' ':>{y_width - 1}} ┼" + "─" * (col_width * n)
    lines.append(x_line)

    x_labels = " " * y_width + " "
    step = max(1, len(buckets) // (bar_area // max(3, len(buckets[0]) + 1))) if bar_area > 0 else 1
    for i in range(n):
        lbl = buckets[i] if i % step == 0 else ""
        x_labels += f"{lbl:<{col_width}}"
    lines.append(x_labels)

    text = Text()
    for i, line in enumerate(lines):
        if i > 0:
            text.append("\n")
        text.append(line)
    return text


def _nice_range_ticks(lo: float, hi: float, num_ticks: int = 5) -> list[float]:
    """Compute nice ticks for an arbitrary range [lo, hi]."""
    span = hi - lo
    if span <= 0:
        return [lo, lo + 1]
    raw_step = span / (num_ticks - 1)
    magnitude = 10 ** math.floor(math.log10(raw_step))
    residual = raw_step / magnitude
    if residual <= 1.0:
        nice_step = magnitude
    elif residual <= 2.0:
        nice_step = 2 * magnitude
    elif residual <= 5.0:
        nice_step = 5 * magnitude
    else:
        nice_step = 10 * magnitude
    start = math.floor(lo / nice_step) * nice_step
    ticks = []
    v = start
    while v <= hi + nice_step * 0.5:
        ticks.append(v)
        v += nice_step
    return ticks


def _draw_line_segment(
    grid: list[list[str]], x0: int, y0: int, x1: int, y1: int
) -> None:
    """Draw a line between two points on the grid using ─ / \\ characters."""
    dx = x1 - x0
    if dx == 0:
        return
    for x in range(x0, x1 + 1):
        t = (x - x0) / dx
        y = y0 + t * (y1 - y0)
        yi = int(round(y))
        if 0 <= yi < len(grid) and 0 <= x < len(grid[0]):
            if grid[yi][x] == " ":
                if y1 > y0:
                    grid[yi][x] = "/"
                elif y1 < y0:
                    grid[yi][x] = "\\"
                else:
                    grid[yi][x] = "─"
```

def render_dashboard(
    *,
    title: str,
    summary: dict[str, Any],
    timeseries: list[dict[str, Any]],
    all_buckets: list[str],
    chart_width: int = 0,
) -> Text:
    """Compose the full TUI dashboard: header + 3 charts."""
    if chart_width <= 0:
        chart_width = os.get_terminal_size(fallback=(80, 24)).columns - 2

    tz_label = _local_tz_label()

    # Build lookup from timeseries
    ts_map = {row["bucket"]: row for row in timeseries}

    calls = [ts_map.get(b, {}).get("calls", 0) for b in all_buckets]
    input_tok = [
        ts_map.get(b, {}).get("input_tokens", 0)
        + ts_map.get(b, {}).get("cache_read_tokens", 0)
        + ts_map.get(b, {}).get("cache_create_tokens", 0)
        for b in all_buckets
    ]
    output_tok = [ts_map.get(b, {}).get("output_tokens", 0) for b in all_buckets]

    # Cache hit per bucket
    cache_hit: list[float | None] = []
    for b in all_buckets:
        row = ts_map.get(b)
        if row is None or row["calls"] == 0:
            cache_hit.append(None)
        else:
            total_in = row["input_tokens"] + row["cache_read_tokens"] + row["cache_create_tokens"]
            if total_in > 0:
                cache_hit.append(row["cache_read_tokens"] / total_in * 100)
            else:
                cache_hit.append(None)

    # Header
    total_in = summary.get("total_input_tokens", 0)
    cache_pct = (
        f"{summary.get('cache_read_tokens', 0) / total_in * 100:.0f}%"
        if total_in > 0 else "n/a"
    )
    header = (
        f"═══ {title} ({tz_label}) {'═' * max(1, chart_width - len(title) - len(tz_label) - 8)}\n"
        f" Calls: {summary.get('total_calls', 0)}"
        f" │ Input: {_fmt_axis_label(total_in)}"
        f" │ Output: {_fmt_axis_label(summary.get('total_output_tokens', 0))}"
        f" │ Cache hit: {cache_pct}"
        f" │ Avg: {summary.get('avg_elapsed_s', 0):.1f}s"
    )

    result = Text()
    result.append(header)
    result.append("\n\n")

    bar = render_bar_chart(
        buckets=all_buckets, values=calls,
        y_label="calls", chart_height=12, chart_width=chart_width,
    )
    result.append_text(bar)
    result.append("\n\n")

    stacked = render_stacked_bar_chart(
        buckets=all_buckets,
        series_a=input_tok, label_a="input",
        series_b=output_tok, label_b="output",
        y_label="tokens", chart_height=12, chart_width=chart_width,
    )
    result.append_text(stacked)
    result.append("\n\n")

    line = render_line_chart(
        buckets=all_buckets, values=cache_hit,
        y_label="cache hit (%)", chart_height=8, chart_width=chart_width,
    )
    result.append_text(line)

    return result
```

def _fmt_tokens(n: int) -> str:
    if n >= 1_000_000:
        return f"{n / 1_000_000:.1f}M"
    if n >= 1_000:
        return f"{n / 1_000:.1f}K"

    total = data.get("total_input_tokens", 0)
    if total == 0:
        return "n/a"
    return f"{data.get('cache_read_tokens', 0) / total * 100:.0f}%"


def _print_summary(title: str, data: dict[str, Any]) -> None:

    print(f"  Total calls:      {data.get('total_calls', 0)}")
    print(f"  Input tokens:     {_fmt_tokens(data.get('total_input_tokens', 0))}")
    print(f"    Non-cached:     {_fmt_tokens(data.get('input_tokens', 0))}")
    print(f"    Cache read:     {_fmt_tokens(data.get('cache_read_tokens', 0))}")
    print(f"    Cache create:   {_fmt_tokens(data.get('cache_create_tokens', 0))}")
    print(f"  Cache hit rate:   {_cache_hit_pct(data)}")
    print(f"  Output tokens:    {_fmt_tokens(data.get('total_output_tokens', 0))}")

    print(f"  Tool rounds:      {data.get('total_tool_rounds', 0)}")
    print(f"  Errors:           {data.get('error_count', 0)}")
    print(f"  Avg latency:      {data.get('avg_elapsed_s', 0):.1f}s")

        print("  (no data)")

    for row in rows:
        print(
            f"  {row[id_key]!s:<15} {row['calls']:>6} "
            f"{_fmt_tokens(row['total_input']):>10} {_fmt_tokens(row['total_output']):>10}"
        )


async def _run_tui(args: argparse.Namespace) -> None:
    tracker = UsageTracker(db_path=_DB_PATH)
    await tracker.init()
    tz_offset = _local_tz_offset_hours()

    try:
        period = args.tui_period

        if period == "day":
            date = args.date or datetime.now(timezone.utc).astimezone().strftime("%Y-%m-%d")
            summary = await tracker.summary_today()
            ts = await tracker.timeseries(period="day", date=date, tz_offset_hours=tz_offset)
            all_buckets = [f"{h:02d}" for h in range(24)]
            title = date

        elif period == "month":
            date = args.date or datetime.now(timezone.utc).astimezone().strftime("%Y-%m")
            summary = await tracker.summary_month(date)
            ts = await tracker.timeseries(period="month", date=date, tz_offset_hours=tz_offset)
            year, month = map(int, date.split("-"))
            num_days = calendar.monthrange(year, month)[1]
            all_buckets = [f"{d:02d}" for d in range(1, num_days + 1)]
            title = date

        elif period == "week":
            date = args.date or datetime.now(timezone.utc).astimezone().strftime("%Y-%m-%d")
            ts = await tracker.timeseries(period="week", date=date, tz_offset_hours=tz_offset)
            # Build 7-day bucket list
            from datetime import timedelta
            end = datetime.strptime(date, "%Y-%m-%d")
            all_buckets = [(end - timedelta(days=6-i)).strftime("%m-%d") for i in range(7)]
            # Summary for the week — use raw query
            rows = await tracker.query_raw(
                """
                SELECT
                    COUNT(*) AS total_calls,
                    COALESCE(SUM(input_tokens), 0) AS input_tokens,
                    COALESCE(SUM(cache_read_tokens), 0) AS cache_read_tokens,
                    COALESCE(SUM(cache_create_tokens), 0) AS cache_create_tokens,
                    COALESCE(SUM(input_tokens + cache_read_tokens + cache_create_tokens), 0) AS total_input_tokens,
                    COALESCE(SUM(output_tokens), 0) AS total_output_tokens,
                    COALESCE(AVG(elapsed_s), 0) AS avg_elapsed_s
                FROM llm_calls

                """,

        )
